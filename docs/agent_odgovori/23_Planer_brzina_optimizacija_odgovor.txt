```python
def _cell_matches_bit(cmask: list[list[int]], r: int, c: int, bit: int, size: int, square: bool) -> bool:
    row = cmask[r]
    if ((c >= 2 and (row[c - 1] & bit) and (row[c - 2] & bit))
            or (1 <= c < size - 1 and (row[c - 1] & bit) and (row[c + 1] & bit))
            or (c + 2 < size and (row[c + 1] & bit) and (row[c + 2] & bit))):
        return True
    if ((r >= 2 and (cmask[r - 1][c] & bit) and (cmask[r - 2][c] & bit))
            or (1 <= r < size - 1 and (cmask[r - 1][c] & bit) and (cmask[r + 1][c] & bit))
            or (r + 2 < size and (cmask[r + 1][c] & bit) and (cmask[r + 2][c] & bit))):
        return True
    if square:
        if r > 0:
            up = cmask[r - 1]
            if c > 0 and (up[c] & bit) and (row[c - 1] & bit) and (up[c - 1] & bit):
                return True
            if c + 1 < size and (up[c] & bit) and (row[c + 1] & bit) and (up[c + 1] & bit):
                return True
        if r + 1 < size:
            dn = cmask[r + 1]
            if c > 0 and (dn[c] & bit) and (row[c - 1] & bit) and (dn[c - 1] & bit):
                return True
            if c + 1 < size and (dn[c] & bit) and (row[c + 1] & bit) and (dn[c + 1] & bit):
                return True
    return False
```

```python
def _line_runs_fast(rows: list[list[int]], non_matchable) -> list[tuple[int, set[tuple[int, int]]]]:
    h = len(rows)
    w = len(rows[0]) if h else 0
    runs: list[tuple[int, set[tuple[int, int]]]] = []
    if w >= 3:
        for r in range(h):
            row = rows[r]
            c = 0
            while c <= w - 3:
                c1 = row[c + 1]
                if c1 != row[c + 2]:
                    c += 2
                    continue
                if row[c] == c1 and c1 > 0 and c1 not in non_matchable:
                    end = c + 3
                    while end < w and row[end] == c1:
                        end += 1
                    runs.append((c1, {(r, cc) for cc in range(c, end)}))
                    c = end
                else:
                    c += 1
    if h >= 3:
        for c in range(w):
            r = 0
            while r <= h - 3:
                c1 = rows[r + 1][c]
                if c1 != rows[r + 2][c]:
                    r += 2
                    continue
                if rows[r][c] == c1 and c1 > 0 and c1 not in non_matchable:
                    end = r + 3
                    while end < h and rows[end][c] == c1:
                        end += 1
                    runs.append((c1, {(rr, c) for rr in range(r, end)}))
                    r = end
                else:
                    r += 1
    return runs


def _square_runs_fast(rows: list[list[int]], non_matchable) -> list[tuple[int, set[tuple[int, int]]]]:
    h = len(rows)
    w = len(rows[0]) if h else 0
    runs: list[tuple[int, set[tuple[int, int]]]] = []
    for r in range(h - 1):
        row0, row1 = rows[r], rows[r + 1]
        for c in range(w - 1):
            color = row0[c]
            if (color > 0 and color == row0[c + 1] and color == row1[c]
                    and color == row1[c + 1] and color not in non_matchable):
                runs.append((color, {(r, c), (r, c + 1), (r + 1, c), (r + 1, c + 1)}))
    return runs
```

### 3. Bitmaske u `engine_refill.fill` / `retry()` + eliminacija `copy.deepcopy` i ponovljenog `decode_dictionary`/`color_maps` (`engine_refill.py` + `engine_spawn_predicates.py`)

**Sta se racuna vise puta i moze da se sredi bez promene RNG-a:**
1. **`retry()` u `fill()` je zvao `combo_count_at` DVAPUT** za istu rupu kad su `flags["special_combo"]` i `flags["combo"]` oba `True` i `combo_count_at == 3` (`is_part_of_combo` samo zove `combo_count_at > 0`).
2. **Susedne celije oko rupe `(x, y)` se NE menjaju tokom retry petlje za tu rupu.** Samo `(x, y)` menja boju! Umesto 9.3 miliona `dict.get` poziva kroz `slots` i 32 patterna u `is_part_of_possible_highrank_combo(slots, x, y, 1/2)` na svaki neuspesan pokusaj:
   - Na pocetku `fill()` izgradimo dve `rows x columns` celobrojne mreze bitmaski `cmask` (`matches_with_color`) i `fmask` (`_fits`, tj. `_matchable and matches_with_color`).
   - Za rupu `(x, y)` se **jednom** po rupi (samo za flagove koji su `True`!) izracunaju celobrojne maske mogucih boja:
     - `axis_mask`: bitmaska boja koje prave horizontalnu/vertikalnu trojku kroz `(x, y)` (koristi `cmask`, kao `_axis_combo`).
     - `sq_raw` / `sq_mask = sq_raw & ~axis_mask`: bitmaska boja koje prave `2x2` kvadrat kroz `(x, y)` (koristi `fmask`, kao `is_part_of_square`; `& ~axis_mask` cuva **tacnu** logiku `combo_count_at` koji vraca `3`, a ne `4`, ako postoji osni kombo!).
     - `any_combo_mask = axis_mask | sq_raw`.
     - `hr1_mask`, `hr2_mask`: OR preko predracunatih trojki ofseta `(ox1,oy1, ox2,oy2, ox3,oy3)` bez `(0,0)` iz `highrank_patterns(1/2)`.
   - Tada ceo `retry()` postaje **4 bitne operacije** nad `bit = 1 << candidate`!
   - Za rupu upisujemo `slots[y][x] = {...}` **jednom** na kraju `while` petlje (ne pri svakom pokusaju), i azuriramo `cmask[y][x] = fmask[y][x] = bit` za naredne rupe.
3. **`copy.deepcopy(board), copy.deepcopy(special)` u `fill_simulator_board`:** zameniti sa `[row[:] for row in board]` (ili `.tolist()` za `np.ndarray`). I u `fill_simulator_board` -> `model.fill(raw, ..., _reuse_raw=True)` izbeci ponovno kloniranje svih 64 dict-ova kad je `raw` tek napravljen.
4. **Kesiranje `decode_dictionary` i `color_maps(self.profile)` na `EngineRefill`:** u `fill()`, `_spawn_special_tiles()`, `_spawn_special_tile()` i `fill_simulator_board()`, ove funkcije se sada zovu po ~4 800–25 000 puta iako se `profile` ne menja (osim kad `booster_energy_weight` zameni `profile`/`spawning` novim dict-om).

```python
@lru_cache(maxsize=2)
def highrank_neighbor_triples(rank):
    """Svaki pattern iz highrank_patterns(rank) bez (0, 0), kao (ox1, oy1, ox2, oy2, ox3, oy3)."""
    triples = []
    for pattern in highrank_patterns(rank):
        others = [pt for pt in pattern if pt != (0, 0)]
        if len(others) == 3:
            (x1, y1), (x2, y2), (x3, y3) = others
            triples.append((x1, y1, x2, y2, x3, y3))
    return tuple(triples)
```

```python
def _slot_color_masks(slot):
    """(color_mask, fit_mask) za EngineRefill.fill() — identicno matches_with_color i _fits."""
    color = int(slot.get("color", 0))
    if color in (0, 1):
        return 0, 0
    value = int(slot.get("value", 0))
    if not (value == 0 or 4 <= value <= 8 or 11 <= value <= 13 or 17 <= value <= 19
            or (23 <= value < 34 and ((0x221800000 >> value) & 1))

        cm = -1

        if c2 in (0, 1):
            cm = 1 << color
        elif c2 == 12:
            cm = -1
        else:
            cm = (1 << color) | (1 << c2)
    fm = cm if slot.get("canBeMatched", color > 1) else 0
    return cm, fm
```

```python
                        if hr2_mask is None:
                            hr2_mask = _hr_mask(triples_2)
                        if not (hr2_mask & bit):
                            break
                    else:
                        break
                    color = priority[index]
                    index += 1
                    attempts.append(color)

                win_bit = 1 << color
                cmask[y][x] = win_bit
                fmask[y][x] = win_bit
                slots[y][x] = {"color": color, "color2": 1, "value": 0, "originalValue": 0,
                               "canBeMatched": True, "movable": True, "score": 1, "energy": 1}
```


### 4. Smanjenje overhead-a u `strategy_lab/simulation.py`: `step_move`, `rules()`, `engine_refill()`, `unknown` sken i kasni `model.clone()`

Sta trosi ~1.7–2.4 s na 38 976 poziva (`step_move` 0.68 s + `rules` 0.75 s + `contextlib.__enter__` 0.96 s):
1. **`rules(move)`:** kad je `key == _ACTIVE_RULES` i nema slojeva na tabli, trenutno ipak pravi `ExitStack()` i **uvek ulazi u `ms.jelly_zone_cells(())`** kad je `JELLY_EXACT_PORTED=True`!
   Resenje: pratiti `_ACTIVE_JELLY_NONEMPTY` u `simulation.py`; ako je `key == _ACTIVE_RULES` i nijedan board sloj nije aktivan (niti je prethodno `jelly_zone_cells` bio neprazan), uraditi samo `yield; return` bez `ExitStack`!
2. **`step_move`:**
   - Premestiti `UnsupportedRefill`, `simulator_refill`, `simulate_swipe`, `has_possible_move`, `shuffle_board` na vrh modula umesto `from ... import ...` unutar svake petlje.

   - **`model = model.clone()`:** pomeriti tek **posle** provera za `PREFIX_MODES`, `unknown`, `jelly/funky/hot_color`, `foxy_powerup`, `billie`, `rainbow lotus`, `perks` i (za `swap` bez `mixy`/`broom`/`cleo`) `can_swap_pair` — tako preostali odbaceni specijal-potezi ne kloniraju 624-elementni `EngineRNG`.

### 5. Preskakanje simetricnog suprotnog poteza (`B -> A`) u `planner._plan()` (`strategy_lab/planner.py`)

**Odgovor na pitanje 4 (dedup pre `step`-a):**
U `actions(state)`, za svaki par `(A, B)` emituju se oba smera `A -> B` i `B -> A` zato sto polozaj novog specijala (`4+`, `T/L`, `2x2`), `UFO` redosled, `Froggy lotus` (`dragged_cell`) i `Disco Ball` (`start` bonus) zavise od smera prevlacenja. Za obicnu plain-plain trojku, oba smera daju identicno stanje i `B -> A` se tek **posle** celog `step()` poziva odbaci u `if key in seen: continue`.

Umesto pokusaja da to pogodimo pre `step`-a, posle uspesnog `step(line.state, action)` za plain-plain `A -> B` (`kind in ("swap", "energy_boost")`) vec imamo `result["events"]`. Ako su ispunjeni svi uslovi da `B -> A` **ne moze** da se razlikuje:
- oba polja su plain (`special_board[r1][c1] == 0 and special_board[r2][c2] == 0`)
- `not line.state.get("every_match_is_special")`
- `getattr(line.state.get("engine_refill"), "mixy", None) is None`
- nijedan dogadjaj u `result["events"]` nije `ufo_color_lock`, `froggy_lotus_gather`, niti mec u kaskadi 1 sa `tile_count >= 4` (gde se pravi specijal na `target` celiji)
onda mozemo dodati obrnuti potpis `(kind, target, start)` u `skip_reverse: set` za taj `line` i preskociti `B -> A` **bez poziva `step()`**!

**Fajl:** `strategy_lab/planner.py` — `_plan()`
```python

            no_every_spec = not line.state.get("every_match_is_special")
            no_mixy = getattr(line.state.get("engine_refill"), "mixy", None) is None
            skip_reverse = set()

                           else actions(line.state)):

                        e.get("type") in ("ufo_color_lock", "froggy_lotus_gather")
                        or (e.get("type") == "match" and int(e.get("cascade") or 0) == 1
```

## MENJA REZULTAT (odlucuje korisnik posebno)

### A. `PLAN_WORKERS > 1` za prvi plan (`strategy_lab/planner.py` + `square_solo_live.py`)

**Vazna analiza `_plan_split` koda u `strategy_lab/planner.py`:**
1. **Ne ubrzava fiksnu sirinu `width`, vec siri pretragu `P` puta!** U `_plan_root`, svaki radnik dobije `1/P` korenskih poteza, ali za sve naredne korake (`index >= 1`) i dalje drzi **pun beam sirine `width`**. Dakle za `P=4`, `width=25` radi **4x vise ukupnog posla** (`4 x 25`), pa je po zidu tek ~1.2–1.5x brze od jednog jezgra (ustedi se samo korak 0 + malo manji `diverse`), ali daje **bolju liniju** u istom roku.
2. **OPASNOST ako se postavi globalno `PLAN_WORKERS > 1`:** `square_solo_live._background_pool()` pokrece `background_plan` u `ProcessPoolExecutor` daemon procesu! Ako `background_plan` pozove `plan()` sa `PLAN_WORKERS > 1`, `_split_pool()` ce pokusati da napravi `Pool` iz daemon procesa i puci sa `AssertionError: daemonic processes are not allowed to have children` (sto `_take_background` tiho proguta kao `"pozadinski plan nije stigao"`).
3. **Cena pickle-a stanja:** `state` sadrzi `EngineRefill` + `profile` (~30–100 KB ugnjezdenih dict-ova) + 624-int `rng.matrix`. Za `P=4` radnika: ~4–8 ms slanje + ~3–5 ms vracanje `SPLIT_LINES=30` linija. Zanemarljivo u odnosu na `width >= 10` (1–15 s), **ali** prvo pravljenje pool-a na Windows-u (`spawn` + import `numpy` i 30k linija `move_strategy`) kosta **~1.5–3 s** jednokratno — ako nije pre-warm-ovano pri startu meca, prvi plan ce izgubiti te 2 s. `[PRETPOSTAVKA]`

**Predlog ako zelis jaci prvi plan (uz ostavljanje 2 jezgra emulatoru + 1 za pozadinski proces):**
- U `_plan_split` dodati `import multiprocessing` i guard `if multiprocessing.current_process().daemon:` -> padni nazad na obican `_plan` (da `background_plan` nikad ne pukne).
- Postaviti `planner.PLAN_WORKERS = min(4, max(1, (os.cpu_count() or 2) - 2))` samo uzivo u `square_solo_live` (sa pre-warm-om `_split_pool()` na ulasku u mec), tek kad zelis siri kvalitet pretrage.
- **Menja rezultat:** **DA** (efektivno `P * width` korensko grananje daje drugu/bolju liniju nego `width` na jednom jezgru).

### B. Pozadinski plan: progresivno sirenje + `stop_early=True` (`square_solo_live.py`)

**Zasto ne vredi nastavljati stari beam i zasto sadasnji skoro uvek puca:**
1. **Stari beam:** posle `ahead = 4..10` poteza (`_start_background`), 90%+ grana iz proslog beama ima drugaciji prefiks akcija i leze na drugoj tabli/RNG-u, pa se stari beam ne moze iskoristiti osim sto `carry` sluzi kao `baseline`.
2. **Zasto uvek udara u rok (`"prolaz prekinut"`):**
   - `_start_background` pokrece **samo jednu sirinu** (`start_width=BG_MAX_WIDTH=25, max_width=25`). Kad je tabla gusca od one na kojoj je izmeren `bg_pace`, `25` prekoraci `15 s`, preseca se na pola dubine i zadnje korake igra na `width=1` — sto je cesto losije od kompletnog `width=10`!
   - `_start_race_background` prosledjuje `seconds = 3 * 3.9 * 0.7 = 8.19 s`, ali **bez `stop_early=True`**. Kad `width=10` traje ~3.5 s, on ipak zapocne `width=25` (~11 s), zakasni na `finish_cut` rep posle `8.19 s`, i kad se odigra 3. potez `_race_take_background` vidi `future.done() == False` i **baci ceo rad u djubre**!

**Predlog u `square_solo_live.py`:**
- U `_start_background`: pokrenuti `start_width=10, max_width=BG_MAX_WIDTH (25)` i dodati `stop_early=True` u `planner.background_plan` / `plan_anytime` (nakon ubrzanja 1–5, `width=10` staje za ~1–2 s a `width=25` za ~3–6 s pa ce se oba zavrsavati bez prekida; na najtezim tablama bar ostaje ceo `width=10` umesto isecenog `25`).
- U `_start_race_background`: dodati `stop_early=True` (i smanjiti `RACE_BG_SHARE` sa `0.7` na npr. `0.55–0.6` da `finish_cut` rep uvek stigne pre `future.done()` provere).
- **Menja rezultat:** **DA** (u zivom rezimu moze izabrati kompletnu `width=10` liniju kad `width=25` ne stize u rok, i ranije preuzimati poboljsanja u trci). `[PRETPOSTAVKA]`
